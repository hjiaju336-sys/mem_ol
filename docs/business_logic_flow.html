mem_ol 业务逻辑流程图

单轮对话、召回、后台维护、回退；世界导入见第二张。

1. 单轮对话 + 召回 + 后台维护

flowchart TD
    U["玩家输入"] --> PREP["LangGraph prepare
计算 turn_no / 读取最近 5 轮"]

    subgraph TURN["单轮主链路（同步）"]
        PREP --> A2A["API2-A：单轮总结 + 查询改写
只看上一轮"]
        PREP --> A2B["API2-B：实体假设
最近 5 轮"]
        A2A --> CTXQ["上下文 query
rewritten_query + context_summary"]
        A2B --> ENTQ["实体检索词 entities"]
        ENTQ --> EK["Neo4j 关键词召回
CJK 全文索引"]
        ENTQ --> EV["PG entity_index 向量召回"]
        EK --> FUSE["entity_id 去重 + RRF"]
        EV --> FUSE
        FUSE --> EXP["1~2 跳图扩展
节点/边/环路去重"]
        CTXQ --> CR["子块向量召回
父块展开 + 父块去重"]
        EXP --> RR["bge-reranker 精排
实体 0.6 / 上下文 0.4"]
        CR --> RR
        RR --> GM["GM 生成回复
最近 5 轮原文 + 主线 + 召回"]
    end

    GM --> PERSIST["persist
消息落库 + 本轮 L0 摘要暂存"]
    PERSIST --> REPLY["返回玩家"]

    subgraph BG["后台任务（BackgroundTasks，顺序受 ingest_lock 约束）"]
        API3["API3 实体/属性/关系固化"]
        ING["5 轮增量向量入库"]
        PLOTM["主线维护
L0 归档 / 生成 L1 / 更新 L2"]
    end

    PERSIST --> API3
    PERSIST --> ING
    PERSIST --> PLOTM

    subgraph STORE["存储"]
        PG[("PostgreSQL + pgvector
messages / chunks / entity_index
retrieval_traces / plot_nodes")]
        NEO[("Neo4j
Entity / AttributeChange / REL")]
    end

    API3 --> NEO
    API3 --> PG
    ING --> PG
    PERSIST --> L0["L0 单轮摘要"]
    PLOTM --> L0
    PLOTM --> L1["L1 章节摘要"]
    PLOTM --> L2["L2 全局梗概"]
    L2 --> GM
    L1 --> GM

    subgraph RB["回退 / 重试"]
        UI["前端：最近 5 轮
回退 / 重试"]
        RS["RollbackService
mark rolled_back + 物理删除"]
    end

    UI -.-> RS
    RS --> PG
    RS --> NEO
    RS --> L0
    RS --> L1
    RS --> L2

2. 世界导入（初始 + 增量）

flowchart LR
    WT["世界文本
doc_id / char_start / char_end"] --> SPLIT["按 20000 字切窗"]
    SPLIT --> PAR["6 路并行抽取
单窗超时 240s；失败二分"]
    WT --> VEC["父子段切分 + 向量化
只嵌入子块"]
    PAR --> CONS["整理 agent
按 30000 字符树形合并"]
    CONS --> MATCH{"会话已有实体？"}
    MATCH -- 否 --> CREATE["entity_commit_plan
create"]
    MATCH -- 是 / append --> UPDATE["规范名/别名匹配
update + 新增"]
    CREATE --> NEO[("Neo4j")]
    UPDATE --> NEO
    CREATE --> PG[("PG entity_index")]
    UPDATE --> PG
    VEC --> PG
