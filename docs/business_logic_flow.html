1. 单轮对话 + 分流 + 召回 + 状态 + 后台维护

flowchart TD
    U["玩家输入 / ClientContext
custom_prompt + pending + lorebook"] --> CLASSIFY["输入分流 API2-A

    CLASSIFY -- meta --> META["元交互
OOC 回答 / 临时指令
不落库、不计 turn"]
    CLASSIFY -- narrative / mixed --> PREP["LangGraph prepare
计算 turn_no / 清空上一轮字段"]

    subgraph TURN["单轮主链路（同步）"]
        PREP --> A2A["API2-A：单轮总结 + 查询改写"]
        PREP --> A2B["API2-B：实体假设
最近 5 轮"]
        A2A --> CTXQ["上下文 query
rewritten_query + context_summary"]
        A2B --> ENTQ["实体检索词 entities"]
        ENTQ --> EK["Neo4j 关键词召回
CJK 全文索引"]
        ENTQ --> EV["PG entity_index 向量召回"]
        EK --> FUSE["entity_id 去重 + RRF"]
        EV --> FUSE
        FUSE --> EXP["1~2 跳图扩展
节点/边/环路去重"]
        CTXQ --> CR["子块向量召回
父块展开 + 父块去重"]
        EXP --> RR["bge-reranker 精排
实体 0.6 / 上下文 0.4"]
        CR --> RR
        RR --> STATE["state 节点
读 game_states

        STATE --> GM["GM 五段提示词

+ 世界书 + 记忆
状态卡 + 本轮结算"]
        GM --> RECON["reconcile
GM effects 校验

        RECON --> PERSIST["persist

    end

    PERSIST --> REPLY["返回玩家"]
    PERSIST --> BG

2. 世界导入（初始 + 增量）

flowchart LR
    WT["世界文本
doc_id / char_start / char_end"] --> SPLIT["按窗口切分
默认 20000 字"]
    SPLIT --> PAR["N 路并行抽取
默认 6；超时/失败二分"]
    WT --> VEC["父子段切分 + 向量化
只嵌入子块"]
    PAR --> CONS["整理 agent
树形合并 / 别名 / 自由关系标签"]
    CONS --> EVENT["重要约定/事件实体化
婚约 / 三年之约 / 拜师 / 契约"]
    EVENT --> MATCH{"会话已有实体？"}
    MATCH -- 否 --> CREATE["entity_commit_plan
create"]
    MATCH -- 是 / append --> UPDATE["规范名/别名匹配
update + 新增"]
    CREATE --> NEO[("Neo4j
Entity / REL.relation_label")]
    UPDATE --> NEO
    CREATE --> PG[("PG entity_index")]
    UPDATE --> PG
    VEC --> PG
